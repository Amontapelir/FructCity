# FructCity — бэкенд на FastAPI (ТЗ 12.1, 12.2)
#
# Здесь указаны НИЖНИЕ ГРАНИЦЫ, а не точные версии. Причина: проект
# живёт на Python 3.14, он вышел недавно, и часть пакетов ещё выкладывает
# сборки под него. Жёсткий пин на версию без готового «колеса» заставляет
# pip компилировать пакет из исходников — для pydantic-core это сборка
# на Rust, которая без установленного компилятора просто падает.
#
# Границы выбраны по первому выпуску с поддержкой 3.14.
#
# ПОСЛЕ успешной установки версии надо зафиксировать — иначе через месяц
# на другой машине встанет другой набор:
#
#     pip freeze > backend/requirements.lock.txt
#
# Файл с точными версиями кладётся в репозиторий, и на сервере ставится
# именно он. Этот файл остаётся для обновлений.

fastapi>=0.118
uvicorn>=0.34
pydantic>=2.12
pydantic-settings>=2.8

# Работа с БД. psycopg 3 вместо psycopg2: активнее развивается и раньше
# выкладывает сборки под новые версии Python.
SQLAlchemy>=2.0.44
alembic>=1.16
psycopg[binary]>=3.3.4

# Тесты. Прогоняются штатным unittest: `python -m unittest discover

#
# httpx нужен для запросов к приложению через ASGITransport, то есть
# без сокета и без порта. `fastapi.testclient.TestClient` не берём:

# об этом при каждом прогоне.
httpx>=0.28.1

# pytest пока не используется — оставлен на случай, когда понадобятся
# фикстуры для слоя БД. Если так и не понадобится, строку убрать.
pytest>=8.4

# Намеренно НЕ включены:
#   uvicorn[standard] — набор extras тянет httptools, watchfiles и uvloop.
#     uvloop под Windows не ставится вовсе, остальные два — с бинарной
#     сборкой, то есть ещё две возможные точки отказа на свежем Python.
#     Для разработки хватает обычного uvicorn; extras вернём на сервере,
#     где Linux и стабильная версия Python.
#   passlib / bcrypt  — пароли хешируются hashlib.scrypt из стандартной
#     библиотеки, форматом с параметрами внутри строки хеша. Меньше
#     зависимостей в слое безопасности — меньше поводов срочно обновляться.
#   python-jose        — сессии на подписанных cookie через hmac,
#     JWT в этом проекте не нужен.
#   tzdata             — приходит сама вместе с psycopg под Windows;
#     календарь в calc.py умеет работать и без неё, на фиксированном
#     смещении Москвы.
